חזרה לבחירה

בדרך האבות והאמהות

הַנְּסִיעָה שֶׁל הָרַבִּי הָרָיָי"צ לְקֶבֶר רָחֵל וּלְחֶבְרוֹן

בְּקַיִץ תרפ"ט עָרַךְ הָרַבִּי הָרָיָי"צ אֶת מַסָּעוֹ הַמְּפֻרְסָם לְאַרְצֵנוּ הַקְּדוֹשָׁה. וְכָךְ כּוֹתֵב הָרַבִּי בַּמִּכְתָּב:

"רָגִיל הָיִיתִי בִּהְיוֹתִי בְּאֶרֶץ מוֹלַדְתִּי, לְבַקֵּר מִזְּמַן לִזְמַן בְּהֵיכְלֵי קֹדֶשׁ צִיּוּנֵי אָבוֹת קְדוֹשִׁים, הוֹד כ"ק אֲבוֹתֵינוּ רַבּוֹתֵינוּ הק' זצוקללה"ה נבג"מ זי"ע, לָשֵׂאת רִנָּה וּתְפִלָּה בְּהִתְעוֹרְרוּת רַחֲמִים רַבִּים בַּעֲדֵנוּ וּבְעַד כָּל תַּלְמִידֵינוּ ואנ"ש בְּתוֹךְ כְּלַל אַחֵינוּ בֵּית יִשְׂרָאֵל שי'. בַּזְּמַן הַזֶּה הִנְנִי מֻשְׁלָל הַיְּכֹלֶת לִנְסֹעַ לְאֶרֶץ מוֹלַדְתִּי, וְהֶחְלַטְתִּי בְּעֶזְרָתוֹ יִת' לַעֲשׂוֹת מַסָּעִי זֶה לְאֶרֶץ הַקֹּדֶשׁ תובב"א, לְבַקֵּר בִּמְקוֹמוֹת הַקֹּדֶשׁ".

וְכָךְ מְסֻפָּר:

יוֹם שְׁלִישִׁי, ז' מְנַחֵם אָב, בְּשָׁעָה 12.45 בְּצָהֳרַיִם נָסַע הָרַבִּי מִיְּרוּשָׁלַיִם לְחֶבְרוֹן.

תַּהֲלוּכָה שְׁלֵמָה שֶׁל עֶשְׂרוֹת אוֹטוֹבּוּסִים עִם כַּמָּה מֵאוֹת אִישׁ, כָּל תַּלְמִידֵי יְשִׁיבַת "תּוֹרַת אֱמֶת", הַתְּמִימִים וְזִקְנֵי אנ"ש לִוְּתָה אֶת רַבֵּנוּ בְּדַרְכּוֹ.

בַּדֶּרֶךְ לְחֶבְרוֹן סָר לְקֶבֶר רָחֵל אִמֵּנוּ, חָקַר לְסֵדֶר הַהַשְׁכָּבָה כְּדֵי לְהִתְפַּלֵּל לְצַד הָרֹאשׁ, עַל פָּנָיו נִכֶּרֶת הַדְרַת קֹדֶשׁ וְהִתְרוֹמְמוּת הַנֶּפֶשׁ, הִתְפַּלֵּל וּבָכָה הַרְבֵּה, הִקִּיף אֶת הַקֶּבֶר, קָרָא שׁוּב אֶת הַ"פִּדְיוֹנוֹת" וְהִדְלִיק ח"י נֵרוֹת, מִשָּׁם נָסַע לְחֶבְרוֹן.

לְיַד הַמּוֹשָׁב הַחֲרֵדִי הֶחָדָשׁ "מִגְדַּל עֵדֶר" יָצְאוּ לְקַבֵּל אֶת פָּנָיו מֵחֶבְרוֹן חַכְמֵי הָעִיר רַבָּנֶיהָ וְנִכְבָּדֶיהָ מֵעֲדַת הַסְּפָרַדִּים וּמֵעֲדַת הָאַשְׁכְּנַזִּים. רַבֵּנוּ מָסַר לכאו"א שָׁלוֹם וּבְרָכָה.

הַבִּקּוּר בְּחֶבְרוֹן

הַקָּדוֹשׁ ר' אֱלִיעֶזֶר דָּן סְלָנִים הי"ד תִּכְנֵן אֶת בִּקּוּרוֹ שֶׁל רַבֵּנוּ בְּחֶבְרוֹן. כַּיָּדוּעַ הָיְתָה הַכְּנִיסָה לְתוֹךְ הַבִּנְיָן הָעוֹמֵד עַל מְעָרַת הַמַּכְפֵּלָה אֲסוּרָה לַיְּהוּדִים, אוּלָם ר' אֱלִיעֶזֶר דָּן סְלָאנִים — שֶׁהָיוּ לוֹ קְשָׁרִים טוֹבִים בְּחוּגֵי הַשִּׁלְטוֹן הָעַרְבִי שֶׁבְּחֶבְרוֹן, הִצְלִיחַ לְהַשִּׂיג רִשָּׁיוֹן מְיֻחָד עֲבוּר רַבֵּנוּ וּשְׁלוֹשָׁה מְלַוִּים. רַבֵּנוּ הִגִּיעַ לְחֶבְרוֹן בְּשָׁעָה 3:30, וְסָר לְבֵיתוֹ שֶׁל הָרַב יַעֲקֹב יוֹסֵף סְלָנִים רַב הָעִיר.

בְּשָׁעָה 5:00 הָלַךְ לִמְעָרַת הַמַּכְפֵּלָה בְּלִוְיַת כָּל תּוֹשָׁבֵי הָעִיר — אֲנָשִׁים וְנָשִׁים — וְכָל בְּנֵי הַיְּשִׁיבָה, הַשַּׁיָּרָה עָבְרָה דֶּרֶךְ הַכְּבִישׁ הַחִיצוֹנִי, לֹא דֶּרֶךְ הַשּׁוּק וְהַחֲנוּיוֹת. רַבֵּנוּ הִתְפַּלֵּל מִנְחָה בִּבְכִיָּה רַבָּה — בְּעֵרֶךְ כִּשְׁתֵּי שָׁעוֹת — וּבְכִיּוֹת עֲצוּמוֹת נִשְׁמְעוּ מִכָּל עֵבֶר וּפִנָּה. הַתָּכְנִית הָיְתָה לְהִכָּנֵס לִמְעָרַת הַמַּכְפֵּלָה דֶּרֶךְ "שַׁעַר יַעֲקֹב", אֲשֶׁר בּוֹ הָיְתָה הַכְּנִיסָה אֲסוּרָה לִיהוּדִים לַחֲלוּטִין (אֲפִלּוּ לְתוֹךְ הַשַּׁעַר), וְלָצֵאת דֶּרֶךְ "שַׁעַר אַבְרָהָם" אֲשֶׁר שָׁם הָיוּ הַמַּדְרֵגוֹת שֶׁלַּיְּהוּדִים הָיְתָה הָרְשׁוּת לְהִכָּנֵס וְלַעֲלוֹת עַד שֶׁבַע מַדְרֵגוֹת בִּלְבַד.

בְּשָׁעָה 7:00 נִכְנַס רַבֵּנוּ בִּפְנִים. הַכְּנִיסָה הָיְתָה דֶּרֶךְ "שַׁעַר יַעֲקֹב", שָׁם חִכְּתָה מִשְׁלַחַת מִגְּדוֹלֵי הַסּוֹחֲרִים הָעַרְבִיִּים שֶׁבְּחֶבְרוֹן אֲשֶׁר בָּאוּ לִכְבוֹדוֹ שֶׁל ר' אֱלִיעֶזֶר דָּן סְלָנִים שֶׁהָיָה אָז מְנַהֵל הַבַּנְק אַנְגְּלוֹ פָּלֶשְׂתִּינָה בע"מ בְּחֶבְרוֹן. הָעַרְבִים קִבְּלוּ אֶת פְּנֵי רַבֵּנוּ וּמְלַוָּיו — חֲתָנוֹ הרש"ג, הָרַב יַעֲקֹב יוֹסֵף סְלָנִים ר' אֱלִיעֶזֶר דָּן סְלָנִים, ר' יִשְׂרָאֵל ור' יִצְחָק דְּבוֹרְץ ור' שְׁלֹמֹה זַלְמָן קְלוֹנְסְקִי — בִּבְרָכָה, וְנָתְנוּ לְכֻלָּם סַנְדְּלֵי עוֹר מְיֻחָדִים לְקָשְׁרָם מֵעַל הַנַּעֲלַיִם, בִּכְדֵי שֶׁלֹּא יֻצְרְכוּ לְהָסִיר הַנַּעֲלַיִם וּלְהִכָּנֵס יְחֵפִים כְּדֶרֶךְ שֶׁהֵם עוֹשִׂים.

רַבֵּנוּ נִכְנַס לִמְקוֹם קִבְרֵי הָאָבוֹת, וְעָבַר לְיַד הַמַּצֵּבוֹת הַגְּדוֹלוֹת שֶׁהָיוּ מְכֻסּוֹת בִּשְׁטִיחִים יְקָרִים וּבְפָרוֹכוֹת רְקוּמוֹת הַפְּרָחִים וַעֲטוּרוֹת בִּגְדִילֵי זָהָב. הַסּוֹחֲרִים הָעַרְבִיִּים הִסְבִּירוּ כְּפִי הַמָּסֹרֶת שֶׁבְּיָדָם, כָּל מַצֵּבָה וּמַצֵּבָה עַל קִבְרוֹ שֶׁל מִי מֵהַזּוּגוֹת הִיא עוֹמֶדֶת. לְיַד מַצֵּבָתָם שֶׁל אַבְרָהָם וְשָׂרָה הֶרְאוּ פֶּתַח שֶׁל בּוֹר הַמּוֹבִיל לַמְּעָרָה וּבְתוֹכוֹ תְּלוּיָה מְנוֹרַת שֶׁמֶן זַיִת הַדּוֹלֶקֶת כְּנֵר תָּמִיד.

רַבֵּנוּ הָלַךְ בָּרֹאשׁ וְכֻלּוֹ שָׁקוּעַ בְּמַחֲשָׁבוֹת וּבְהִרְהוּרִים, הָיָה מֻפְלָא לִרְאוֹת אֵיךְ שֶׁרַבֵּנוּ מוֹצֵא אֶת דַּרְכּוֹ בִּסְבַךְ הַדֶּרֶךְ כִּי הָיָה נִדְמֶה כָּל הָעֵת שֶׁעֵינָיו עֲצוּמוֹת וּשְׂפָתָיו נָעוֹת וּמְמַלְמְלוֹת. כָּךְ עָבַר הַבִּקּוּר בַּמְּעָרָה, עַד שֶׁהִגִּיעַ לְ"שַׁעַר אַבְרָהָם" [אֲשֶׁר שָׁם הָיוּ הַמַּדְרֵגוֹת שֶׁלַּיְּהוּדִים הָיְתָה הָרְשׁוּת לְהִכָּנֵס לְשַׁעַר זֶה], רַבֵּנוּ יָרַד בַּמַּדְרֵגוֹת הָרַבּוֹת עַד שֶׁהִגִּיעַ לַמַּדְרֵגָה הַשְּׁבִיעִית, אֲשֶׁר שָׁם [מִמַּדְרֵגָה הַשְּׁבִיעִית וְעַד הַחוּץ] חִכָּה לְרַבֵּנוּ כָּל הַקָּהָל.

מִשָּׁם הָלַךְ לְסַיֵּר אֶת נַחֲלָתוֹ וַחֲצֵרוֹ, הִתְעַנְיֵן בְּכָל הַפְּרָטִים, מְקוֹמוֹ, מַעֲמָדוֹ וּמַצָּבוֹ שֶׁל הַבַּיִת וּמִגְרָשָׁיו שֶׁסְּבִיבוֹ. סָר שׁוּב לְבֵית הָרַב יַעֲקֹב יוֹסֵף סְלָנִים, נָח מְעַט וְהָלַךְ לְבֵית הַכְּנֶסֶת הַגָּדוֹל שֶׁהָיָה מָלֵא מִפֶּה אֶל פֶּה וְאָמַר מַאֲמָר יְהִי ה' אֱלֹקֵינוּ עִמָּנוּ.

עַל פִּי בַּקָּשַׁת הַנְהָלַת יְשִׁיבַת סְלוֹבוֹדְקָה שֶׁבְּחֶבְרוֹן הָלַךְ לְבַקֵּר אֶת הַיְּשִׁיבָה.

בְּשָׁעָה 10 בַּלַּיְלָה שָׁב לִירוּשָׁלַיִם.

רָחֵל מֵאִמָּא רָחֵל

סִפּוּרָהּ שֶׁל הגב' רָחֵל וֶרֶד רָם, הַמְּבַקֶּרֶת מִדֵּי יוֹם אֵצֶל רָחֵל אִמֵּנוּ וּבְאַמְתַּחְתָּהּ רְשִׁימָה עֲמוּסָה שֶׁל שֵׁמוֹת עֲלֵיהֶם הִיא מַעְתִּירָה בִּתְפִלָּה.

רָחֵל אִמֵּנוּ. הָאִמָּא הָאַגָּדִית עִם סִפּוּרֵי הַוִּתּוּר הַמְּיֻחָדִים הַמְּהַוִּים הַשְׁרָאָה עַד יָמֵינוּ אֵלֶּה.

יַעֲקֹב רָצָה לִקְבֹּר אֶת רָחֵל בִּמְעָרַת הַמַּכְפֵּלָה, אַךְ רָחֵל וִתְּרָה – כְּאִמָּא אֲמִתִּית נִשְׁאֲרָה עַל אֵם הַדֶּרֶךְ כְּדֵי לִהְיוֹת לְעֵזֶר לְבָנֶיהָ בְּשָׁעָה שֶׁיֵּצְאוּ לְגָלוּת. רַק הִיא מִכֻּלָּם קִבְּלָה אֶת הַבְטָחַת הַגְּאֻלָּה.

נְשָׁמָה מְאִירָה

"נוֹלַדְתִּי וְגָדַלְתִּי בְּבַיִת מָסָרְתִּי, בּוֹ הָיוּ עוֹשִׂים קִדּוּשׁ אַךְ לֹא מַמָּשׁ שׁוֹמְרִים שַׁבָּת" – מְסַפֶּרֶת רָחֵל וֶרֶד רָם מִיְּרוּשָׁלַיִם, אוֹ בְּכִנּוּיָהּ – "רָחֵל מֵאִמָּא רָחֵל".

"הוֹרַי שָׁלְחוּ אוֹתִי לְבֵית סֵפֶר דָּתִי כְּדֵי שֶׁ'נֵדַע קְצָת עַל אֱלוֹקִים'. בְּגִיל שְׁמוֹנֶה הָלַכְתִּי לִקְבוּצַת תְּהִלִּים, שָׁם אָמְרוּ לָנוּ שֶׁאָסוּר לִנְסֹעַ בְּשַׁבָּת. חָזַרְתִּי הַבַּיְתָה וְאָמַרְתִּי לְאִמִּי שֶׁיּוֹתֵר לֹא אֶסַּע אִתָּם לַעֲשׂוֹת פִּיקְנִיק כְּהֶרְגֵּלֵנוּ בְּשַׁבָּתוֹת. מֵאָז הָיִיתִי נִשְׁאֶרֶת, שַׁבָּת אַחֲרֵי שַׁבָּת, בְּכָל מִינֵי מְקוֹמוֹת, אֲפִלּוּ אֵצֶל אֲנָשִׁים שֶׁלֹּא הִכַּרְתִּי. כְּבָר אָז נִצְנְצָה בִּי נַפְשִׁי הָאֱלוֹקִית. כַּיּוֹם אֲנִי מְקֹרֶבֶת מְאוֹד לָרַבִּי".

הַקֶּשֶׁר עִם רָחֵל

"הַבֵּן שֶׁלִּי נִפְצַע חֹדֶשׁ לִפְנֵי הַשִּׁחְרוּר מֵהַצָּבָא וְהָיָה חַיָּב לַעֲבֹר נִתּוּחַ" מַמְשִׁיכָה רָחֵל בְּסִפּוּרָהּ. "יוֹם לִפְנֵי הַנִּתּוּחַ הִתְהַלַּכְתִּי כְּסַהֲרוּרִית וּמְפֻחֶדֶת. דָּאַגְתִּי לוֹ מְאוֹד. לְפֶתַע קָפְצָה לְמֹחִי הַמַּחְשָׁבָה 'הֲרֵי אֲנִי גָּרָה יַחֲסִית קָרוֹב, נֵלֵךְ לְהִתְפַּלֵּל אֵצֶל רָחֵל אִמֵּנוּ'. בָּכִיתִי, בִּקַּשְׁתִּי מִמֶּנָּה שֶׁתִּשְׁמֹר עַל הַבֵּן שֶׁלִּי וְתַעֲזֹר לוֹ בַּנִּתּוּחַ.

"הִרְגַּשְׁתִּי שֶׁהִיא אִתִּי, אִמָּא אֲמִתִּית – מַקְשִׁיבָה, שׁוֹמַעַת וְעוֹזֶרֶת. וְאָכֵן הַנִּתּוּחַ עָבַר בְּהַצְלָחָה רַבָּה. מֵאָז הִתְחַלְתִּי לְהַרְגִּישׁ קֶשֶׁר מְיֻחָד לְרָחֵל אִמֵּנוּ. מֵאָז אֲנִי ׳הַשְּׁלִיחָה׳ שֶׁל קֶבֶר רָחֵל. לִפְעָמִים מַגִּיעָה לְשַׁחֲרִית, מִנְחָה, עַרְבִית – נוֹסַעַת הָלוֹךְ וָשׁוֹב עִם הָרֶכֶב".

שְׁלִיחוּת בְּקֶבֶר רָחֵל

"לְקֶבֶר רָחֵל מַגִּיעִים אֲנָשִׁים רַבִּים לְתַנּוֹת אֶת צָרוֹתֵיהֶם. כְּ'וָתִיקָה' בַּמָּקוֹם, אֲנִי מַרְגִּישָׁה שֶׁאֲנִי יְכוֹלָה לְסַיֵּעַ לָהֶם", מְשַׁתֶּפֶת רָחֵל. "אֲנִי עוֹזֶרֶת, מַקְשִׁיבָה וְעוֹשָׂה כָּל מָה שֶׁאֲנִי יְכוֹלָה. בְּכָל בִּקּוּר בְּקֶבֶר רָחֵל יֵשׁ הַרְבֵּה עֲבוֹדָה…".

הַסִּפּוּרִים רַבִּים מִסְּפֹר, אֲצַיֵּן כַּמָּה לְדֻגְמָא:

בְּעֶרֶב פּוּרִים רָצִיתִי לְשַׁלֵּם עַל הַזְּכוּת שֶׁעֲלִיָּה מְסֻיֶּמֶת בַּתּוֹרָה תִּהְיֶה שֶׁלִּי. זֶה הָיָה חָשׁוּב לִי מְאוֹד. אֲנִי וְאִשָּׁה נוֹסֶפֶת לֹא הָיִינוּ מוּכָנוֹת לְוַתֵּר, וְהִתְמוֹדַדְנוּ… הִיא מַעֲלָה אֶת הַסְּכוּם וַאֲנִי מַעֲלָה… וְאָז הִיא אוֹמֶרֶת לִי: ׳אָנָּא, וַתְּרִי לִי. הַבֵּן שֶׁלִּי בַּכֶּלֶא, אֲנִי רוֹצָה שֶׁיֵּצֵא מִשָּׁם׳. לֹא רַק שֶׁוִּתַּרְתִּי לָהּ, אֶלָּא הִתְחַלְתִּי גַּם לְהִתְפַּלֵּל עָלָיו אֵצֶל רָחֵל. וְאָכֵן הִתְרַחֵשׁ הַנֵּס, וב"ה, הוּא שֻׁחְרַר מֻקְדָּם מֵהַצָּפוּי!

"וְהִנֵּה סִפּוּר נוֹסָף: מִישֶׁהִי, רַוָּקָה מְבֻגֶּרֶת שֶׁלֹּא מָצְאָה אֶת זִוּוּגָהּ, תָּרְמָה 350 ₪ לְצֹרֶךְ הַשְׂכָּרַת מָקוֹם בְּקֶבֶר רָחֵל עֲבוּר הִתְוַעֲדוּת קְבוּעָה שֶׁאֲנִי עוֹרֶכֶת בַּמָּקוֹם. כַּמּוּבָן שֶׁהִתְפַּלַּלְתִּי עָלֶיהָ רַבּוֹת. כַּעֲבֹר תְּקוּפָה בִּקְּשָׁה לִתְרֹם שׁוּב, לַמְרוֹת שֶׁהִיא הָיְתָה חַסְרַת אֶמְצָעִים, אַךְ הִיא הִתְעַקְּשָׁה. בָּכִיתִי לְאִמָּא רָחֵל. ׳תִּרְאִי כַּמָּה הִיא עוֹשָׂה בִּשְׁבִילֵךְ!'. וב"ה, קָרָה נֵס. כַּעֲבֹר חֳדָשִׁים סְפוּרִים הִתְקַשְּׁרָה לְהוֹדִיעַ שֶׁהִשְׁתַּדְּכָה.

שָׁבוּעַ לִפְנֵי הַהִתְוַעֲדוּת הַקְּבוּעָה, הִתְקַשְּׁרָה אֵלַי הַכַּלָּה. בֵּרַרְתִּי בַּעֲדִינוּת וְנוֹכַחְתִּי לָדַעַת שֶׁאֵין לָהֶם כֶּסֶף אֲפִלּוּ לְאֵרוּעַ הַ׳וָארְט׳. אָמַרְתִּי לָהּ שֶׁנַּעֲשֶׂה אֶת הַ'וָארְט' בַּהִתְוַעֲדוּת הַקְּבוּעָה בְּעוֹד שָׁבוּעַ. הֵפַצְתִּי בִּקְבוּצוֹת שֶׁאֵין כְּנִיסָה בְּלִי מַתָּנָה לַכַּלָּה… וְהִיא אָכֵן קִבְּלָה נְדוּנְיָה כְּיַד הַמֶּלֶךְ. רָחֵל אִמֵּנוּ הֶחְזִירָה לָהּ בְּגָדוֹל!"…

הָרַבִּי נוֹכֵחַ בְּחַיַּי

אֵיךְ נוֹצַר הַקֶּשֶׁר שֶׁלָּךְ עִם הָרַבִּי?

"לִפְנֵי כַּמָּה שָׁנִים רָצִיתִי לְהִתְחַזֵּק, וְלֹא יָדַעְתִּי אֵיךְ. שָׁמַעְתִּי שֶׁהַשְּׁלִיחָה שֶׁל הָרַבִּי אֶצְלֵנוּ בַּשְּׁכוּנָה עוֹרֶכֶת הִתְוַעֲדוּת. הִגַּעְתִּי, וּמֵאָז הִתְחַלְתִּי לְהִשְׁתַּתֵּף בַּהִתְוַעֲדוּיוֹת וּבַשִּׁעוּרִים שֶׁל חַבָּ"ד. מֵאָז אֲנִי מַרְגִּישָׁה שֶׁהָרַבִּי מְלַוֶּה אוֹתִי כָּל הַזְּמַן, נוֹכֵחַ בְּחַיַּי, וְדוֹאֵג לִי כְּמוֹ אַבָּא.

זָכִיתִי גַּם לִהְיוֹת כַּמָּה פְּעָמִים אֵצֶל הָרַבִּי. בְּכָל פַּעַם הִרְגַּשְׁתִּי מֵחָדָשׁ אֵיךְ הָרַבִּי דּוֹאֵג לִי גַּשְׁמִית וְרוּחָנִית. מַמָּשׁ שׁוֹמֵר עָלַי וּמְעוֹדֵד אוֹתִי".

ביאור של הרבי על מעלת רחל אימנו

תכונת הוויתור של עקרת הבית

על הכתוב בפ' ויחי "ואני בבואי מפדן מתה עלי רחל גו׳ ואקברה שם בדרך", מפרש רש"י "ע"פ הדיבור קברתי' שם, שתהא לעזרה לבני', כשיגלה אותם נבוזראדן כו' יוצאת רחל על קברה ובוכה ומבקשת עליהם רחמים כו', והקב"ה משיבה יש שכר לפעולתך".

ויש לבאר הטעם שרחל דווקא נקברה בדרך כדי "שתהא לעזרה לבני'", ולא יעקב אבינו בעצמו:

מצינו שרחל אימנו "היא הייתה עקרת הבית" (פרש"י ויצא לא, ד). וכן מובא במדרש (בראשית רבה פע"א, ב): "רחל הייתה עיקרו של בית".

והנה מצינו בנוגע לנשי ישראל, שפטורות ממ"ע שהזמן גרמא (משנה קידושין כט, א), וכן פטורות מתלמוד תורה (קידושין שם, ב). ומבואר בראשונים (אבודרהם סדר תפלות של חול. כלבו ר"ס ע"ג) שהטעם לזה הוא מפני שעל הנשים מוטלים צורכי הבית, ולכן פטורות מדברים אלו.

ונמצא, שלנשי ישראל יש תכונה נפלאה של ויתור – שכדי שיוכלו לשמש בתפקידן כ"עקרת הבית", הרי הן מוותרות על עבודת השי"ת שיש בה קדושה גלוי', כמו לימוד התורה וקיום מ"ע, ובמקום זה עובדות את ה' על ידי העיסוק בצורכי הבית.

ועפ"ז יש לבאר הטעם שרחל אימנו נקברה בדרך ולא יעקב אבינו, כי מאחר שרחל הייתה "עקרת הבית", לכן הייתה בה מעלה זו - לוותר במשך מאות ואלפי שנים על הקבורה במקום של קדושה גלוי', מערת המכפלה, כדי שכשיהיו ישראל במעמד ומצב בלתי רצוי, עד כדי כך שיתחייבו גלות ‑ "תהא לעזרה לבני'!".

(ע"פ לקוטי שיחות ח"ל עמ' 239 ואילך. אור החסידות)

חברון, עיר האבות

הַמְּעָרָה הָעַתִּיקָה

אֵבֶל גָּדוֹל יָרַד עַל הָעִיר חֶבְרוֹן וּסְבִיבָתָהּ. שָׂרָה, הַצַּדֶּקֶת הַגְּדוֹלָה, הָאִשָּׁה שֶׁעָשְׂתָה כָּל כָּךְ הַרְבֵּה מַעֲשֵׂי חֶסֶד, וְעָזְרָה לְכָל אָדָם בִּמְאוֹר פָּנִים, נִפְטְרָה.

"אַתָּה יָכוֹל לִקְבֹּר אֶת שָׂרָה בְּקִבְרֵי הַמְּלָכִים שֶׁלָּנוּ, בַּמָּקוֹם הַמְּכֻבָּד בְּיוֹתֵר", אָמְרוּ רָאשֵׁי הָעִיר לְאַבְרָהָם.

אֲבָל אַבְרָהָם יָדַע שֶׁהוּא רָצָה לִקְבֹּר אֶת שָׂרָה בְּמָקוֹם אַחֵר.

זֶה קָרָה 38 שָׁנִים קֹדֶם לָכֵן. שְׁלוֹשָׁה מַלְאָכִים בָּאוּ לְבַקֵּר אֶת אַבְרָהָם, וּבְאוֹתוֹ בִּקּוּר הוֹדִיעוּ לוֹ כִּי בְּעוֹד שָׁנָה יִוָּלֵד לוֹ וּלְשָׂרָה בֶּן – יִצְחָק. לִכְבוֹד בּוֹאָם שֶׁל הָאוֹרְחִים הָלַךְ אַבְרָהָם אֶל הָעֵדֶר כְּדֵי לָקַחַת עֵגֶל לְהָכִין מִמֶּנּוּ אֲרוּחָה מְכֻבֶּדֶת. הוּא הִתְחִיל לָלֶכֶת עִם הָעֵגֶל, וּפִתְאוֹם הָעֵגֶל הֵחֵל לִבְרֹחַ. אַבְרָהָם רָץ אַחֲרָיו, וְהָעֵגֶל הִתְרַחֵק עוֹד וָעוֹד הוּא עָלָה עַל גְּבָעוֹת וְעָבַר בֵּין עֵצִים, עַד שֶׁנִּכְנַס לְתוֹךְ מְעָרָה.

אַבְרָהָם נִכְנַס אֶל הַמְּעָרָה בְּעִקְבוֹת הָעֵגֶל וְעָמַד מִשְׁתָּאֶה. רֵיחַ נִפְלָא מִלֵּא אֶת הַמָּקוֹם. אַבְרָהָם מִסְתַּכֵּל פְּנִימָה וְרָאָה שְׁתֵּי מִטּוֹת, וַעֲלֵיהֶן יְשֵׁנִים אִישׁ וְאִשָּׁה זְקֵנִים מְאוֹד. לְיַד רָאשֵׁיהֶם דָּלְקוּ נֵרוֹת.

אַבְרָהָם רָאָה אֶת הַמַּחֲזֶה הַמֻּפְלָא וְעָמַד עַל מְקוֹמוֹ בְּחֶרְדַּת קֹדֶשׁ. פִּתְאוֹם שָׁמַע קוֹל מִמְּרוֹמִים. זֶה הָיָה קוֹלָם שֶׁל מַלְאֲכֵי הַשָּׁרֵת. הַמַּלְאָכִים אָמְרוּ "זֶה הַקֶּבֶר שֶׁל אָדָם וְחַוָּה, וּבֶעָתִיד יִקְבְּרוּ בַּמְּעָרָה הַזֹּאת גַּם אֶת אַבְרָהָם וְשָׂרָה, יִצְחָק וְרִבְקָה, יַעֲקֹב וְלֵאָה".

לָקַח מִשָּׁם אַבְרָהָם אֶת הָעֵגֶל, וְעַכְשָׁו הָלַךְ הָעֵגֶל בְּשֶׁקֶט וְלֹא נִסָּה לִבְרֹחַ. אַבְרָהָם הֵבִין שֶׁהָעֵגֶל בָּרַח רַק כְּדֵי לְהַרְאוֹת לוֹ אֶת הַמְּעָרָה הַסּוֹדִית.

כְּשֶׁנִּפְטְרָה שָׂרָה יָדַע אַבְרָהָם שֶׁהוּא רוֹצֶה לִקְבֹּר אֶת שָׂרָה שָׁם בְּאוֹתָהּ מְעָרָה עַתִּיקָה. הוּא גַּם לֹא רָצָה לְקַבֵּל אוֹתָהּ בְּמַתָּנָה, אֶלָּא הִתְעַקֵּשׁ לְשַׁלֵּם תְּמוּרָתָהּ אֶת הַמְּחִיר הַמָּלֵא.

אַבְרָהָם קָנָה אֶת כָּל הַשָּׂדֶה שֶׁהַמְּעָרָה שׁוֹכֶנֶת בּוֹ, וּבַמְּעָרָה הַזֹּאת הוּא קָבַר אֶת שָׂרָה אִשְׁתּוֹ. כְּשֶׁהוּא נִפְטַר - קָבְרוּ אוֹתוֹ לְיָדָהּ. בַּהֶמְשֵׁךְ נִקְבְּרוּ בְּאוֹתָהּ מְעָרָה גַּם יִצְחָק וְרִבְקָה, יַעֲקֹב וְלֵאָה. וְעַד הַיּוֹם הַזֶּה בָּאִים יְהוּדִים לְהִתְפַּלֵּל בִּמְקוֹם הַקֹּדֶשׁ הַזֶּה - בִּמְעָרַת הַמַּכְפֵּלָה. מתוך הספר 'החומש שלי'.

ביאור של הרבי על מעלת העיר חברון

"עניין האחדות נמצא בהדגשה יתרה בחברון, מאחר ששם קבורים האבות והאימהות של כל בני ישראל, ולכן התפילות עולות על ידי חברון. ותפילה נקראת 'חברון' מצד התחברות . . הקישור והחיבור של יהודים עם הקדוש ברוך הוא . .

ומאחר שהם האבות והאימהות של כלל ישראל, אזי באמצעות החיבור של האבות והאימהות של כלל ישראל ביחד, נעשה גם חיבור אמיתי של הבנים והבנות שמגיעים מהם.

ועניין החיבור הוא כל כך מודגש, שזה מתבטא בגלוי 'בשמו אשר יקראו לו בלשון הקודש', בשם העיר 'חברון', שזה מלשון חיבור".

ושם בהערה 69:

ב'מגלה עמוקות' אופן רמ"ט איתא הגהה מבן המחבר, ושם: "ולכן איתא ב'זוהר חדש', דרך מערת המכפלה נכנס התפילה למעלה".

וזה לשונו ב'מגלה עמוקות' שם: ". . וכן כל תפילות שלנו כשהם עולים למעלה, עולים דרך מערת המכפלה".

"כידוע שכל התפילות צריכות לעלות למעלה דרך פתח של גן עדן הארץ, ועל כן התפילה שמתפללים אצל מערת המכפלה – היא רצויה ומקובלת ממש, לפי שיש בקבלה בידינו ששם היה פתחו של גן עדן ממש".

'לקוטי שיחות' חלק כ"ה עמוד 98.
